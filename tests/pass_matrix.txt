{
  int[3][4] m;
  int i;
  int j;
  int trace;

  i = 0;
  while (i < 3) {
    j = 0;
    while (j < 4) {
      m[i][j] = i * 4 + j;
      j = j + 1;
    }
    i = i + 1;
  }
  print(m[2][3]);
  print(m[1][0] + m[0][1]);

  trace = 0;
  i = 0;
  while (i < 3) {
    trace = trace + m[i][i];
    i = i + 1;
  }
  print(trace);
}
