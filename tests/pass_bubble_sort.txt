{
  int[6] v;
  int n;
  int i;
  boolean swapped;

  n = 6;
  v[0] = 42; v[1] = -7; v[2] = 13; v[3] = 0; v[4] = 99; v[5] = 13;

  do {
    swapped = false;
    i = 1;
    while (i < n) {
      if (v[i - 1] > v[i]) {
        int tmp;
        tmp = v[i - 1];
        v[i - 1] = v[i];
        v[i] = tmp;
        swapped = true;
      }
      i = i + 1;
    }
    n = n - 1;
  } while (swapped);

  i = 0;
  while (i < 6) {
    print(v[i]);
    i = i + 1;
  }
}
